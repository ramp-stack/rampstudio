================================================================================
  CONCEPT
================================================================================

RampStudio is a code editing IDE built on Quartz. It composes three external
crates — editor, explorer, terminal — into a single fullscreen application
with a resizable panel layout. Two layout modes are supported: side-by-side
(explorer | editor | terminal) and stacked (explorer | editor over terminal).
Panel dividers are draggable. Settings persist to a single settings.json file.

The window layout chrome is handled entirely by the window-layout crate.
RampStudio constructs a SetupConfig, passes it to window_layout::setup::mount,
and delegates all mouse handling and per-tick updates to window_layout::logic.

================================================================================
  CANVAS VARS  (written by window-layout, prefixed wl_)
================================================================================

wl_ratio_a          f32   Explorer right edge as fraction of canvas width.
wl_ratio_b          f32   Editor right edge as fraction of canvas width (sbs).
wl_ratio_c          f32   Editor bottom edge as fraction of panel height (stacked).
wl_drag_which       u8    0 = none, 1 = divider A, 2 = divider B/C.
wl_layout_mode      u8    0 = side-by-side, 1 = stacked.
wl_sidebar_active   u8    0 = closed, non-zero = active icon's sidebar_id.
wl_min_explorer     f32   Minimum explorer width. Synced from ExplorerSettings
                          each tick by the update loop in lib.rs.
wl_topbar_h         f32   Topbar height. Set at mount from SetupConfig.
wl_sidebar_w        f32   Sidebar width. Set at mount from SetupConfig.
wl_div_a_hover      u8    1 when mouse is hovering divider A.
wl_div_b_hover      u8    1 when mouse is hovering divider B.
wl_div_c_hover      u8    1 when mouse is hovering divider C.
wl_explorer_opening u8    1 while divider A drag is opening the explorer.

================================================================================
  SETTINGS  (src/settings.rs)
================================================================================

All serialisation helpers (color_to_hex, parse_hex_color, find_val, get_f,
get_b, get_us, get_col, serialize_*, parse_*) are private to this module.

pub fn ensure_file()
  Creates settings.json with default values if it does not already exist.
  Defaults: backspace_deletes_before = true, auto_pairs = true.

pub fn load(ed, ex, term)
  Reads settings.json. Parses all three sections into the provided structs.
  Missing keys left at their struct defaults.

pub fn save(ed, ex, term)
  Writes a single settings.json containing editor, explorer, terminal blocks.

================================================================================
  PROJECT  (src/project.rs)
================================================================================

pub fn resolve_project_root() -> String
  Reads RAMP_PROJECT env var. Canonicalises to absolute path if possible.
  Falls back to "." if unset or empty.

pub fn pick_initial_file(root: &str) -> Option<String>
  Tries in order: src/main.rs, src/lib.rs, main.rs, lib.rs, README.md,
  readme.md, index.js, index.ts. Falls back to first file found by read_dir.
  Returns None if root is empty.

Phosphor Bold glyph codepoints (private):
  PH_FILES / PH_SEARCH / PH_USERS / PH_TERMINAL / PH_GIT / PH_EXTENSIONS

pub fn build_config(icon_font: Arc<Font>) -> SetupConfig
  Constructs a SetupConfig from the constants above and preferences.
  Sets sidebar_icons vec with all six icons.
  initial_active = ICON_ID_FILES.

pub fn setup(cv, cw, ch, cfg)        → window_layout::setup::mount
pub fn handle_press(cv, mx, my, cfg) → window_layout::logic::on_press
pub fn handle_release(cv)            → window_layout::logic::on_release
pub fn handle_move(cv, mx, my)       → window_layout::logic::on_move
pub fn tick(cv, cfg) -> Panels       → window_layout::logic::update

--------------------------------------------------------------------------------
  components/editor.rs

#[derive(Clone)]
pub struct EditorComponent { inner: Arc<Editor> }

  EditorComponent::new(x, y, w, h, code_font, gutter_font,
                       initial_file, theme_bytes, settings) -> Self
    Wraps Editor::new in Arc.

  fn mount(&self, cv)
    Calls inner.mount(cv) and inner.register_callbacks(cv).

  fn set_bounds(&self, x, y, w, h)
    Delegates to inner.set_bounds.

  fn open_file_callback(&self) -> impl Fn(&str) + 'static
    Returns a closure that calls inner.open_file. Used when wiring the
    explorer on_file_open callback without moving EditorComponent.

#[derive(Clone)]
pub struct ExplorerComponent {
    pub settings: Shared<ExplorerSettings>,
    resize_fn:    Arc<dyn Fn(&mut Canvas, f32, f32, f32, f32)>,
}

  ExplorerComponent::new(context, scene, layer_id,
                         font_bold_b, font_reg_b, font_fa_b,
                         settings, on_file_open) -> Self
    Calls explorer::mount, stores component.resize as Arc<dyn Fn>.
    Arc (not Box) used so the struct can derive Clone.

  fn resize(&self, cv, x, y, w, h)
    Calls resize_fn.

================================================================================
  LIB  (src/lib.rs)
================================================================================

Modules declared: constants, preferences, settings, project, window, components.

App::new(context) -> Scene

  Fonts
    font_bold_b   JetBrainsMono-Bold.ttf      → gutter_font
    font_reg_b    JetBrainsMono-Regular.ttf   → code_font, terminal font
    font_fa_b     fa-solid-900.ttf            → explorer icon font
    font_ph_b     Phosphor-Light.ttf          → sidebar icon font (ph_font)

  Theme
    Include cobalt.tmTheme. Fall back to dark-rust.tmTheme if not present.

  Settings
    settings::ensure_file()
    settings::load() → ed_settings, ex_settings, term_settings
    ed_settings: backspace_deletes_before = true, auto_pairs = true

  Layout
    panel_top    = TOPBAR_H + 1.0
    init_split_a = (INIT_CW * INIT_EXPLORER_RATIO).round()
    init_split_b = (INIT_CW * (1.0 - INIT_TERMINAL_RATIO)).round()

  Project
    project_root = project::resolve_project_root()
    initial_file = project::pick_initial_file(&project_root) or "code.txt"

  Window layout
    wl_cfg = Arc::new(window::build_config(ph_font))
    window::setup(cv, INIT_CW, INIT_CH, &wl_cfg)

  Editor
    EditorComponent::new(init_split_a + DIV_W, panel_top,
                         init_split_b - init_split_a - DIV_W,
                         INIT_CH - panel_top,
                         code_font, gutter_font,
                         &initial_file, &theme_bytes, ed_settings)
    editor.mount(cv)

  Explorer
    ex_settings.x            = wl_cfg.sidebar_w
    ex_settings.y            = panel_top
    ex_settings.w            = init_split_a - wl_cfg.sidebar_w
    ex_settings.project_root = project_root.clone()
    ex_settings_shared       = Shared::new(ex_settings)
    ExplorerComponent::new(..., ex_settings_shared, editor.open_file_callback())
    cv.set_var("wl_min_explorer", ex_settings_shared.get().min_width)

  Terminal
    term_settings.offset_x = init_split_b + DIV_W
    term_settings.offset_y = panel_top
    TerminalComponent::new(..., Shared::new(term_settings), Shared::new(project_root))

  Input handlers
    on_mouse_press   → window::handle_press(cv, mx, my, &cfg)
    on_mouse_release → window::handle_release(cv)
    on_mouse_move    → window::handle_move(cv, mx, my)

  Update loop
    Sync wl_min_explorer from ex_share.get().min_width each tick.
    p = window::tick(cv, &cfg)
    ex_x = if p.explorer_visible { p.explorer.0 } else { -9999.0 }
    Update ex_share: x = ex_x, y = p.explorer.1, w = p.explorer.2
    explorer.resize(cv, ex_x, p.explorer.1, p.explorer.2, p.explorer.3)
    editor.set_bounds(p.editor.0, p.editor.1, p.editor.2, p.editor.3)
    Update ts_share: offset_x = p.terminal.0, offset_y = p.terminal.1 + TAB_H
    cv.set_var("_term_panel_y", p.terminal.1)

  ramp::run! { []; |context| App::new(context) }

================================================================================
  MAIN  (src/main.rs)
================================================================================

Always exactly this. Never changes between projects.

  #[cfg(not(target_arch = "wasm32"))]
  fn main() {
      if let Some(path) = std::env::args().nth(1) {
          std::env::set_var("RAMP_PROJECT", path);
      }
      main::maverick_main();
  }
